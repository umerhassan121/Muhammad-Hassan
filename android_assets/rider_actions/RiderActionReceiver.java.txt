package __PKG__;

import android.app.NotificationManager;
import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.webkit.CookieManager;

import androidx.core.app.NotificationCompat;

import org.json.JSONObject;

import java.io.ByteArrayOutputStream;
import java.io.InputStream;
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;

/**
 * Notification ke Accept / Decline (vendor: Accept / Reject) button dabne par: app khole baghair server ko signed token bhejta hai
 * (POST /shop/push/action/ - wahi endpoint jo PWA ka service worker use karta hai).
 */
public class RiderActionReceiver extends BroadcastReceiver {
    static final String BASE = "__BASE__";

    /** Rider ne app mein jo language chuni (cookie erp_lang) - "ur" ya "en". Cookie na mile to "en". */
    static String appLang() {
        try {
            String ck = CookieManager.getInstance().getCookie(BASE);
            if (ck != null) {
                for (String part : ck.split(";")) {
                    String p = part.trim();
                    if (p.startsWith("erp_lang=")) return "ur".equals(p.substring(9).trim()) ? "ur" : "en";
                }
            }
        } catch (Exception e) {
            // ignore
        }
        return "en";
    }

    @Override
    public void onReceive(final Context ctx, final Intent in) {
        final String token = in.getStringExtra("token");
        final String act = in.getStringExtra("act");
        final int nid = in.getIntExtra("nid", 0);
        final String lang = appLang();
        final boolean ur = "ur".equals(lang);
        if (token == null || token.length() == 0) return;
        final Context c = ctx.getApplicationContext();
        final PendingResult pr = goAsync();
        new Thread(new Runnable() {
            @Override
            public void run() {
                boolean reached = false;
                boolean ok = false;
                String msg;
                try {
                    HttpURLConnection h = (HttpURLConnection) new URL(BASE + "/shop/push/action/").openConnection();
                    h.setRequestMethod("POST");
                    h.setConnectTimeout(5000);
                    h.setReadTimeout(6000);
                    h.setDoOutput(true);
                    h.setRequestProperty("Content-Type", "application/json");
                    byte[] body = ("{\"token\":" + JSONObject.quote(token) + ",\"lang\":" + JSONObject.quote(lang) + "}").getBytes("UTF-8");
                    OutputStream os = h.getOutputStream();
                    os.write(body);
                    os.close();
                    int code = h.getResponseCode();
                    InputStream is = code >= 400 ? h.getErrorStream() : h.getInputStream();
                    ByteArrayOutputStream bo = new ByteArrayOutputStream();
                    if (is != null) {
                        byte[] buf = new byte[1024];
                        int n;
                        while ((n = is.read(buf)) > 0) bo.write(buf, 0, n);
                        is.close();
                    }
                    JSONObject j = new JSONObject(bo.toString("UTF-8"));
                    ok = j.optBoolean("ok", false);
                    msg = j.optString("message", ok ? (ur ? "\u06c1\u0648 \u06af\u06cc\u0627\u06d4" : "Done.") : (ur ? "\u06a9\u0686\u06be \u063a\u0644\u0637 \u06c1\u0648 \u06af\u06cc\u0627\u06d4 \u0628\u0631\u0627\u06c1 \u06a9\u0631\u0645 \u0627\u06cc\u067e \u06a9\u06be\u0648\u0644\u06cc\u06ba\u06d4" : "Something went wrong. Please open the app."));
                    reached = true;
                } catch (Exception e) {
                    msg = ur ? "\u0627\u0646\u0679\u0631\u0646\u06cc\u0679 \u0646\u06c1\u06cc\u06ba \u06c1\u06d2\u06d4 \u0628\u0631\u0627\u06c1 \u06a9\u0631\u0645 \u0627\u06cc\u067e \u06a9\u06be\u0648\u0644 \u06a9\u0631 \u062f\u0648\u0628\u0627\u0631\u06c1 \u06a9\u0648\u0634\u0634 \u06a9\u0631\u06cc\u06ba\u06d4" : "No internet. Please open the app and try again.";
                }
                try {
                    NotificationManager nm = (NotificationManager) c.getSystemService(Context.NOTIFICATION_SERVICE);
                    if (nm != null) {
                        RiderMessagingService.ensureChannels(c, nm);
                        if (reached) nm.cancel(nid);   // jawab mil gaya (ok ya order ab available nahi): offer hata do
                        NotificationCompat.Builder b = new NotificationCompat.Builder(c, RiderMessagingService.CH_GENERAL)
                                .setSmallIcon(c.getApplicationInfo().icon)
                                .setContentTitle(ok ? ("accept".equals(act) ? (ur ? "\u0622\u0631\u0688\u0631 \u0642\u0628\u0648\u0644 \u06c1\u0648 \u06af\u06cc\u0627" : "Order accepted") : ("reject".equals(act) ? (ur ? "\u0622\u0631\u0688\u0631 \u0645\u0633\u062a\u0631\u062f \u06a9\u0631 \u062f\u06cc\u0627" : "Order rejected") : (ur ? "\u0622\u0631\u0688\u0631 \u0645\u0633\u062a\u0631\u062f \u06a9\u0631 \u062f\u06cc\u0627" : "Order declined"))) : (ur ? "\u0622\u0631\u0688\u0631" : "Order"))
                                .setContentText(msg)
                                .setStyle(new NotificationCompat.BigTextStyle().bigText(msg))
                                .setAutoCancel(true)
                                .setTimeoutAfter(ok && "accept".equals(act) ? 600000L : 15000L)
                                .setContentIntent(RiderMessagingService.openAppIntent(c, nid + 7));
                        nm.notify(nid + 7, b.build());
                    }
                } catch (Exception e) {
                    // ignore
                }
                pr.finish();
            }
        }).start();
    }
}
